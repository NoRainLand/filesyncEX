// 设备指纹（FNV-1a）生成稳定 ID
function genDeviceId(): string {
  const parts = [
    navigator.userAgent,
    navigator.language,
    screen.width + "x" + screen.height
  ];
  const hash = fnv1a(parts.join("|"));
  return hash.toString(16).padStart(8, "0");
}
